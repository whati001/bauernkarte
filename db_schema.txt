-- DB Schema --

-- Base
store:
id: BIGINT
name: VARCHAR/TEXT
position: postgis.geography
openinghours: VARCHAR/TEXT -- (or something better)
created: DATETIME
created_by: user.id
modified: DATETIME
modified_by: user.id

product:
id: BIGINT
name: VARCHAR/TEXT
description: VARCHAR/TEXT
created: DATETIME
created_by: user.id
modified: DATETIME
modified_by: user.id

store_product:
id: BIGINT
store: store.id
product: product.id
price: FLOAT(2 dezimal)
created: DATETIME
created_by: user.id
modified: DATETIME
modified_by: user.id

rating:
id: BIGINT
store_product: store_product.id
score: enum(UP) -- only UP is allowed
created: DATETIME
created_by: user.id

images:
id: BIGINT
store: store.id
image: BINARY
description: VARCHAR/TEXT
created: DATETIME
created_by: user.id
modified: DATETIME
modified_by: user.id

----
-- User Management
user:
id: BIGINT
name: VARCHAR/TEXT
email: VARCHAR/TEXT
pwd: CHAR(64) -- (sha256; some pwd type)
verified: BOOLEAN
created: DATETIME
modified: DATETIME
